    16→    16→// ─── fetchJSON helper (retry for dev cold-start) ──────────────────────────
    17→    17→
    18→    18→async function fetchJSON<T = any>(url: string, retries = 2): Promise<T> {
    19→    19→  try {
    20→    20→    const res = await fetch(url);
    21→    21→    return await res.json();
    22→    22→  } catch (err) {
    23→    23→    if (retries > 0) {
    24→    24→      await new Promise((r) => setTimeout(r, 800));
    25→    25→      return fetchJSON<T>(url, retries - 1);
    26→    26→    }
    27→    27→    throw err;
    28→    28→  }
    29→    29→}

    81→    81→/**
    82→    82→ * ConsumerHomeV3 — Complete YouTube-style consumer shell.
    83→    83→ * ----------------------------------------------------
    84→    84→ * - Desktop: collapsible left sidebar (rail when collapsed)
    85→    85→ * - Mobile: bottom navigation (5 items, prominent Create button)
    86→    86→ * - Content area renders different pages based on activeNav
    87→    87→ * - Top bar with search + Create button
    88→    88→ */
    89→    89→export function ConsumerHomeV3() {
    90→    90→  const { setView, playExperience, playSparkQueue } = useStudioStore();
    91→    91→  const [home, setHome] = useState<HomeData | null>(null);
    92→    92→  const [loading, setLoading] = useState(true);
    93→    93→  const [sidebarOpen, setSidebarOpen] = useState(true);
    94→    94→  const [activeNav, setActiveNav] = useState<NavId>('home');
    95→    95→  const [searchQuery, setSearchQuery] = useState('');
    96→    96→
    97→    97→  useEffect(() => {
    98→    98→    let cancelled = false;
    99→    99→    (async () => {
   100→   100→      try {
   101→   101→        const d = await fetchJSON<{ home: HomeData }>('/api/consumer-v2/home?userId=demo-user');
   102→   102→        if (!cancelled) setHome(d.home);
   103→   103→      } catch { /* ignore */ }
   104→   104→      if (!cancelled) setLoading(false);
   105→   105→    })();
   106→   106→    return () => { cancelled = true; };
   107→   107→  }, []);
   108→   108→
   109→   109→  const sparks = home?.sparks ?? [];
   110→   110→  const experiences = home?.experiences ?? [];
   111→   111→  const live = home?.live ?? [];
   112→   112→  const highlights = home?.highlights ?? [];
   113→   113→
   114→   114→  // Search filter (applies to cards rendered on home/sparks/games pages)
   115→   115→  const q = searchQuery.trim().toLowerCase();
   116→   116→  const sparksFiltered = sparks.filter((e) =>
   117→   117→    !q || (e.title?.toLowerCase().includes(q) ?? false) || (e.creatorName?.toLowerCase().includes(q) ?? false),
   118→   118→  );
   119→   119→  const experiencesFiltered = experiences.filter((e) =>
   120→   120→    !q || (e.title?.toLowerCase().includes(q) ?? false) || (e.creatorName?.toLowerCase().includes(q) ?? false),
   121→   121→  );
   122→   122→
   123→   123→  const openSpark = (index: number) => {
   124→   124→    if (sparks.length > 0) playSparkQueue(sparks, index);
   125→   125→  };
   126→   126→
   127→   127→  return (
   128→   128→    <div className="min-h-screen flex flex-col bg-background">
   129→   129→      {/* Mobile top bar */}
   130→   130→      <MobileTopBar
   131→   131→        searchQuery={searchQuery}
   132→   132→        setSearchQuery={setSearchQuery}
   133→   133→        onCreate={() => setView('creator-studio')}
   134→   134→      />
   135→   135→
   136→   136→      <div className="flex flex-1 min-h-0">
   137→   137→        {/* ── Desktop Left Sidebar (collapsible) ── */}
   138→   138→        <DesktopSidebar
   139→   139→          open={sidebarOpen}
   140→   140→          onToggle={() => setSidebarOpen(!sidebarOpen)}
   141→   141→          activeNav={activeNav}
   142→   142→          setActiveNav={setActiveNav}
   143→   143→          onNavigate={(v) => setView(v)}
   144→   144→        />
   145→   145→
   146→   146→        {/* ── Main Content ── */}
   147→   147→        <div className="flex-1 min-w-0 flex flex-col">
   148→   148→          {/* Desktop top bar */}
   149→   149→          <DesktopTopBar
   150→   150→            searchQuery={searchQuery}
   151→   151→            setSearchQuery={setSearchQuery}
   152→   152→            onCreate={() => setView('creator-studio')}
   153→   153→          />
   154→   154→
   155→   155→          <main className="flex-1 overflow-y-auto p-4 pb-24 md:pb-6 space-y-6 max-w-7xl w-full mx-auto">
   156→   156→            {loading ? (
   157→   157→              <LoadingState />
   158→   158→            ) : (
   159→   159→              <ContentRouter
   160→   160→                activeNav={activeNav}
   161→   161→                sparks={sparksFiltered}
   162→   162→                experiences={experiencesFiltered}
   163→   163→                live={live}
   164→   164→                highlights={highlights}
   165→   165→                onPlaySpark={openSpark}
   166→   166→                onPlayExperience={playExperience}
   167→   167→                onPlayAllSparks={() => openSpark(0)}
   168→   168→              />
   169→   169→            )}
   170→   170→          </main>
   171→   171→        </div>
   172→   172→      </div>
   173→   173→
   174→   174→      {/* ── Mobile bottom navigation ── */}
   175→   175→      <MobileBottomNav
   176→   176→        activeNav={activeNav}
   177→   177→        setActiveNav={setActiveNav}
   178→   178→        onCreate={() => setView('creator-studio')}
   179→   179→      />
   180→   180→    </div>
   181→   181→  );
   182→   182→}

   186→   186→function ContentRouter({
   187→   187→  activeNav, sparks, experiences, live, highlights,
   188→   188→  onPlaySpark, onPlayExperience, onPlayAllSparks,
   189→   189→}: {
   190→   190→  activeNav: NavId;
   191→   191→  sparks: Experience[];
   192→   192→  experiences: Experience[];
   193→   193→  live: LiveStream[];
   194→   194→  highlights: Highlight[];
   195→   195→  onPlaySpark: (i: number) => void;
   196→   196→  onPlayExperience: (id: string) => void;
   197→   197→  onPlayAllSparks: () => void;
   198→   198→}) {
   199→   199→  switch (activeNav) {
   200→   200→    case 'home':
   201→   201→      return <HomePage sparks={sparks} experiences={experiences} live={live} onPlaySpark={onPlaySpark} onPlayExperience={onPlayExperience} onPlayAllSparks={onPlayAllSparks} />;
   202→   202→    case 'sparks':
   203→   203→      return <SparksPage sparks={sparks} onPlay={onPlaySpark} onPlayAll={onPlayAllSparks} />;
   204→   204→    case 'games':
   205→   205→      return <GamesPage experiences={experiences} onPlay={onPlayExperience} />;
   206→   206→    case 'live':
   207→   207→      return <LivePage live={live} />;
   208→   208→    case 'highlights':
   209→   209→      return <HighlightsPage highlights={highlights} />;
   210→   210→    case 'tournaments':
   211→   211→      return <TournamentsPage />;
   212→   212→    case 'subscriptions':
   213→   213→      return <SubscriptionsPage experiences={experiences} onPlay={onPlayExperience} />;
   214→   214→    case 'library':
   215→   215→      return <LibraryPage />;
   216→   216→    case 'history':
   217→   217→      return <EmptyState icon={History} title="No history yet" subtitle="Experiences you watch will appear here" />;
   218→   218→    case 'liked':
   219→   219→      return <EmptyState icon={Heart} title="No liked content yet" subtitle="Tap the heart on experiences you enjoy" />;
   220→   220→    case 'watch-later':
   221→   221→      return <EmptyState icon={Bookmark} title="Nothing saved yet" subtitle="Bookmark experiences to play them later" />;
   222→   222→    case 'search':
   223→   223→      return <SearchPage />;
   224→   224→    default:
   225→   225→      return <HomePage sparks={sparks} experiences={experiences} live={live} onPlaySpark={onPlaySpark} onPlayExperience={onPlayExperience} onPlayAllSparks={onPlayAllSparks} />;
   226→   226→  }
   227→   227→}

   494→   494→  const [searching, setSearching] = useState(false);

  1099→  1099→  if (!n || n < 0) return '0';
  1100→  1100→  if (n >= 1_000_000) return (n / 1_000_000).toFixed(1).replace('.0', '') + 'M';

  1102→  1102→  return String(n);
  1103→  1103→}
