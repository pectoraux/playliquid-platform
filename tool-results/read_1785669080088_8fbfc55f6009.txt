    16→// ─── fetchJSON helper (retry for dev cold-start) ──────────────────────────
    17→
    18→async function fetchJSON<T = any>(url: string, retries = 2): Promise<T> {
    19→  try {
    20→    const res = await fetch(url);
    21→    return await res.json();
    22→  } catch (err) {
    23→    if (retries > 0) {
    24→      await new Promise((r) => setTimeout(r, 800));
    25→      return fetchJSON<T>(url, retries - 1);
    26→    }
    27→    throw err;
    28→  }
    29→}

    81→/**
    82→ * ConsumerHomeV3 — Complete YouTube-style consumer shell.
    83→ * ----------------------------------------------------
    84→ * - Desktop: collapsible left sidebar (rail when collapsed)
    85→ * - Mobile: bottom navigation (5 items, prominent Create button)
    86→ * - Content area renders different pages based on activeNav
    87→ * - Top bar with search + Create button
    88→ */
    89→export function ConsumerHomeV3() {
    90→  const { setView, playExperience, playSparkQueue } = useStudioStore();
    91→  const [home, setHome] = useState<HomeData | null>(null);
    92→  const [loading, setLoading] = useState(true);
    93→  const [sidebarOpen, setSidebarOpen] = useState(true);
    94→  const [activeNav, setActiveNav] = useState<NavId>('home');
    95→  const [searchQuery, setSearchQuery] = useState('');
    96→
    97→  useEffect(() => {
    98→    let cancelled = false;
    99→    (async () => {
   100→      try {
   101→        const d = await fetchJSON<{ home: HomeData }>('/api/consumer-v2/home?userId=demo-user');
   102→        if (!cancelled) setHome(d.home);
   103→      } catch { /* ignore */ }
   104→      if (!cancelled) setLoading(false);
   105→    })();
   106→    return () => { cancelled = true; };
   107→  }, []);
   108→
   109→  const sparks = home?.sparks ?? [];
   110→  const experiences = home?.experiences ?? [];
   111→  const live = home?.live ?? [];
   112→  const highlights = home?.highlights ?? [];
   113→
   114→  // Search filter (applies to cards rendered on home/sparks/games pages)
   115→  const q = searchQuery.trim().toLowerCase();
   116→  const sparksFiltered = sparks.filter((e) =>
   117→    !q || (e.title?.toLowerCase().includes(q) ?? false) || (e.creatorName?.toLowerCase().includes(q) ?? false),
   118→  );
   119→  const experiencesFiltered = experiences.filter((e) =>
   120→    !q || (e.title?.toLowerCase().includes(q) ?? false) || (e.creatorName?.toLowerCase().includes(q) ?? false),
   121→  );
   122→
   123→  const openSpark = (index: number) => {
   124→    if (sparks.length > 0) playSparkQueue(sparks, index);
   125→  };
   126→
   127→  return (
   128→    <div className="min-h-screen flex flex-col bg-background">
   129→      {/* Mobile top bar */}
   130→      <MobileTopBar
   131→        searchQuery={searchQuery}
   132→        setSearchQuery={setSearchQuery}
   133→        onCreate={() => setView('creator-studio')}
   134→      />
   135→
   136→      <div className="flex flex-1 min-h-0">
   137→        {/* ── Desktop Left Sidebar (collapsible) ── */}
   138→        <DesktopSidebar
   139→          open={sidebarOpen}
   140→          onToggle={() => setSidebarOpen(!sidebarOpen)}
   141→          activeNav={activeNav}
   142→          setActiveNav={setActiveNav}
   143→          onNavigate={(v) => setView(v)}
   144→        />
   145→
   146→        {/* ── Main Content ── */}
   147→        <div className="flex-1 min-w-0 flex flex-col">
   148→          {/* Desktop top bar */}
   149→          <DesktopTopBar
   150→            searchQuery={searchQuery}
   151→            setSearchQuery={setSearchQuery}
   152→            onCreate={() => setView('creator-studio')}
   153→          />
   154→
   155→          <main className="flex-1 overflow-y-auto p-4 pb-24 md:pb-6 space-y-6 max-w-7xl w-full mx-auto">
   156→            {loading ? (
   157→              <LoadingState />
   158→            ) : (
   159→              <ContentRouter
   160→                activeNav={activeNav}
   161→                sparks={sparksFiltered}
   162→                experiences={experiencesFiltered}
   163→                live={live}
   164→                highlights={highlights}
   165→                onPlaySpark={openSpark}
   166→                onPlayExperience={playExperience}
   167→                onPlayAllSparks={() => openSpark(0)}
   168→              />
   169→            )}
   170→          </main>
   171→        </div>
   172→      </div>
   173→
   174→      {/* ── Mobile bottom navigation ── */}
   175→      <MobileBottomNav
   176→        activeNav={activeNav}
   177→        setActiveNav={setActiveNav}
   178→        onCreate={() => setView('creator-studio')}
   179→      />
   180→    </div>
   181→  );
   182→}

   186→function ContentRouter({
   187→  activeNav, sparks, experiences, live, highlights,
   188→  onPlaySpark, onPlayExperience, onPlayAllSparks,
   189→}: {
   190→  activeNav: NavId;
   191→  sparks: Experience[];
   192→  experiences: Experience[];
   193→  live: LiveStream[];
   194→  highlights: Highlight[];
   195→  onPlaySpark: (i: number) => void;
   196→  onPlayExperience: (id: string) => void;
   197→  onPlayAllSparks: () => void;
   198→}) {
   199→  switch (activeNav) {
   200→    case 'home':
   201→      return <HomePage sparks={sparks} experiences={experiences} live={live} onPlaySpark={onPlaySpark} onPlayExperience={onPlayExperience} onPlayAllSparks={onPlayAllSparks} />;
   202→    case 'sparks':
   203→      return <SparksPage sparks={sparks} onPlay={onPlaySpark} onPlayAll={onPlayAllSparks} />;
   204→    case 'games':
   205→      return <GamesPage experiences={experiences} onPlay={onPlayExperience} />;
   206→    case 'live':
   207→      return <LivePage live={live} />;
   208→    case 'highlights':
   209→      return <HighlightsPage highlights={highlights} />;
   210→    case 'tournaments':
   211→      return <TournamentsPage />;
   212→    case 'subscriptions':
   213→      return <SubscriptionsPage experiences={experiences} onPlay={onPlayExperience} />;
   214→    case 'library':
   215→      return <LibraryPage />;
   216→    case 'history':
   217→      return <EmptyState icon={History} title="No history yet" subtitle="Experiences you watch will appear here" />;
   218→    case 'liked':
   219→      return <EmptyState icon={Heart} title="No liked content yet" subtitle="Tap the heart on experiences you enjoy" />;
   220→    case 'watch-later':
   221→      return <EmptyState icon={Bookmark} title="Nothing saved yet" subtitle="Bookmark experiences to play them later" />;
   222→    case 'search':
   223→      return <SearchPage />;
   224→    default:
   225→      return <HomePage sparks={sparks} experiences={experiences} live={live} onPlaySpark={onPlaySpark} onPlayExperience={onPlayExperience} onPlayAllSparks={onPlayAllSparks} />;
   226→  }
   227→}

   491→function SearchPage() {
   492→  const [query, setQuery] = useState('');
   493→  const [results, setResults] = useState<any[]>([]);
   494→  const [searching, setSearching] = useState(false);
   495→  const { playExperience } = useStudioStore();
   496→
   497→  const search = async (q: string) => {
   498→    setQuery(q);
   499→    if (!q.trim()) { setResults([]); return; }
   500→    setSearching(true);
   501→    try {
   502→      const d = await fetchJSON<{ results: any[] }>(`/api/search?q=${encodeURIComponent(q)}`);
   503→      setResults(d.results ?? []);
   504→    } catch { setResults([]); }
   505→    setSearching(false);
   506→  };
   507→
   508→  return (
   509→    <section>
   510→      <SectionHeader icon={<Search className="w-5 h-5 text-muted-foreground" />} title="Search" subtitle="Find experiences, creators, sparks" />
   511→      <div className="flex items-center gap-2 px-3 py-2 rounded-full border border-border bg-muted/50 mb-4">
   512→        <Search className="w-4 h-4 text-muted-foreground" />
   513→        <input value={query} onChange={e => search(e.target.value)} placeholder="Search..." className="flex-1 bg-transparent text-sm focus:outline-none" autoFocus />
   514→        {searching && <Loader2 className="w-4 h-4 animate-spin text-muted-foreground" />}
   515→      </div>
   516→      {results.length > 0 ? (
   517→        <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-3">
   518→          {results.map(r => (
   519→            <div key={r.experienceId} className="rounded-xl overflow-hidden cursor-pointer group hover:bg-muted/30" onClick={() => playExperience(r.experienceId)}>
   520→              <div className="relative aspect-video bg-gradient-to-br from-violet-300 to-fuchsia-300 dark:from-violet-800 dark:to-fuchsia-800 flex items-center justify-center">
   521→                {r.thumbnailUrl ? <img src={r.thumbnailUrl} alt={r.title} className="w-full h-full object-cover" /> : <Gamepad2 className="w-8 h-8 text-white/50" />}
   522→              </div>
   523→              <div className="p-2">

  1096→// ─── Format count (YouTube-style: 1.2K, 3.4M) ──────────────────────────────
  1097→
  1098→function formatCount(n: number): string {
  1099→  if (!n || n < 0) return '0';
  1100→  if (n >= 1_000_000) return (n / 1_000_000).toFixed(1).replace('.0', '') + 'M';
  1101→  if (n >= 1_000) return (n / 1_000).toFixed(1).replace('.0', '') + 'K';
  1102→  return String(n);
  1103→}
